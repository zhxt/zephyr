function(bes_boot_json_string output value)
  string(REPLACE "\\" "\\\\" value "${value}")
  string(REPLACE "\"" "\\\"" value "${value}")
  string(REPLACE "\n" "\\n" value "${value}")
  set(${output} "\"${value}\"" PARENT_SCOPE)
endfunction()

function(bes_boot_unit name source kind flags)
  get_property(index GLOBAL PROPERTY BES_BOOT_UNIT_INDEX)
  if("${index}" STREQUAL "")
    set(index 0)
  endif()
  set(target "boot_unit_${index}")
  math(EXPR next "${index} + 1")
  set_property(GLOBAL PROPERTY BES_BOOT_UNIT_INDEX "${next}")
  if(kind STREQUAL "archive")
    add_library(${target} STATIC "${source}")
  else()
    add_library(${target} OBJECT "${source}")
  endif()
  target_compile_options(${target} PRIVATE ${${flags}})
  if(kind STREQUAL "archive")
    set_target_properties(${target} PROPERTIES OUTPUT_NAME boot_wrappers)
    target_link_libraries(adapter PRIVATE -Wl,--whole-archive ${target} -Wl,--no-whole-archive)
  elseif(kind STREQUAL "memory" OR kind STREQUAL "strlen")
    set(output "${GENERATED_DIR}/${kind}.o")
    add_custom_command(
      OUTPUT "${output}"
      COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/inputs.py"
              --transform "${kind}" --cross-compile "${CROSS_COMPILE}"
              --input $<TARGET_OBJECTS:${target}> --output "${output}"
      DEPENDS ${target} $<TARGET_OBJECTS:${target}> "${CMAKE_CURRENT_SOURCE_DIR}/inputs.py"
      VERBATIM
    )
    set_source_files_properties("${output}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
    target_sources(adapter PRIVATE "${output}")
  else()
    target_sources(adapter PRIVATE $<TARGET_OBJECTS:${target}>)
  endif()
  foreach(key name source kind target)
    bes_boot_json_string(${key}_json "${${key}}")
  endforeach()
  set(options)
  foreach(option IN LISTS ${flags})
    bes_boot_json_string(option_json "${option}")
    list(APPEND options "${option_json}")
  endforeach()
  string(JOIN "," options_json ${options})
  string(CONCAT record
    "{\"name\":${name_json},\"source\":${source_json},"
    "\"kind\":${kind_json},\"target\":${target_json},\"flags\":[${options_json}]}"
  )
  set_property(GLOBAL APPEND PROPERTY BES_BOOT_UNIT_RECORDS "${record}")
endfunction()

include(cmake/flags.cmake)
include(cmake/sources.cmake)
include(cmake/linker.cmake)

get_target_property(link_inputs adapter LINK_DEPENDS)
if(NOT link_inputs)
  message(FATAL_ERROR "Missing loader linker inputs")
endif()
set(link_records)
foreach(path IN LISTS link_inputs)
  bes_boot_json_string(path_json "${path}")
  list(APPEND link_records "${path_json}")
endforeach()
string(JOIN "," link_inputs_json ${link_records})
file(GENERATE OUTPUT "${GENERATED_DIR}/link-inputs.json" CONTENT "[${link_inputs_json}]\n")

get_property(records GLOBAL PROPERTY BES_BOOT_UNIT_RECORDS)
string(JOIN ",\n" records_json ${records})
file(GENERATE OUTPUT "${GENERATED_DIR}/units.json" CONTENT "[\n${records_json}\n]\n")
set(symbols)
foreach(symbol IN LISTS BES_BOOT_REQUIRED_SYMBOLS)
  bes_boot_json_string(symbol_json "${symbol}")
  list(APPEND symbols "${symbol_json}")
endforeach()
string(JOIN "," symbols_json ${symbols})
file(GENERATE OUTPUT "${GENERATED_DIR}/required-symbols.json" CONTENT "[${symbols_json}]\n")
